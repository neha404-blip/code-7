library(pROC)
data<-read.csv("diagnostic_test_data.csv")
head(data)
# Question 1: Confusion Matrix Calculation
TP <- sum(data$disease_status == 1 & data$test_results == 1)
TN <- sum(data$disease_status == 0 & data$test_results == 0)
FP <- sum(data$disease_status == 0 & data$test_results == 1)
FN <- sum(data$disease_status == 1 & data$test_results == 0)
# Print Confusion Matrix Elements
confusion_matrix <- data.frame(
  TP = TP, TN = TN, FP = FP, FN = FN )
print(confusion_matrix)


# Question 2: Sensitivity and Specificity
sensitivity <- TP / (TP + FN)
specificity <- TN / (TN + FP)
# Print Sensitivity and Specificity
cat("Sensitivity:", sensitivity)
cat("Specificity:", specificity)

# Question 3: ROC Curve Analysis

plot(roc_obj)

# Print AUC
cat("AUC:", auc_value)
