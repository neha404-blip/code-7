library(survival)
library(survminer)
study_data<-read.csv("study_data.csv")
head(study_data)
summary(study_data)

table(study_data$treatment)
table(study_data$gender)

cox_model<-coxph(Surv(time,status)~age+treatment+gender,data=study_data)
summary(cox_model)

cox.zph(cox_model)
plot(cox.zph(cox_model))


new_data<-data.frame(

  treatment=c(0,1),
  gender=mean(study_data$gender)
)


install.packages("survminer")

ggsurvplot(surv_fit,data=new_data,
