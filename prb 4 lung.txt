install.packages("survival")
library(survival)
lung=survival::lung
head(lung)

#summary(lung) eta dile shb variable er ta dekhabe

sapply(
  lung[, c("age","wt.loss")], 
       function(x) 
         c(
  Mean = mean(x, na.rm=TRUE),
  Median = median(x, na.rm=TRUE),
  SD = sd(x, na.rm=TRUE),
  Min = min(x, na.rm=TRUE),
  Max = max(x, na.rm=TRUE)
))

#2.


km_fit <- survfit(Surv(time, status == 2) ~ 1, data = lung)
plot(km_fit,
     xlab = "Time (days)",
     ylab = "Survival Probability",
     main = "Kaplan-Meier Survival Curve (Entire Cohort)",
     col = "blue",
     lwd=2
     )



#3.


km_sex <- survfit(Surv(time, status == 2) ~ sex, data = lung)

plot(km_sex,
     col = c("blue", "red"),
     lwd = 2,
     xlab = "Time (days)",
     ylab = "Survival Probability",
     main = "Kaplan-Meier Survival Curve: Male vs Female")
legend("bottomleft",
       legend = c("male", "female"),
       col = c("blue", "red"),
       lwd = 2)

survdiff(Surv(time, status == 2) ~ sex, data = lung)
