### Assumptions of Proportional Hazards (PH)

1. Proportional hazards over time.
2. Independent survival times among individuals.
3. Correct specification of the model.
4. Non-informative censoring.
5. Independence between censoring and survival time.

#####
Exponential Regression Model Assumptions:
1. The hazard rate is constant over time.
2. The survival time follows an exponential distribution.
3. The observations are independent.
4. The censoring is non-informative.
5. The covariates have a multiplicative effect on the hazard.

######Difference from Cox Model:

Exponential model assumes a constant baseline hazard, 
whereas Cox model leaves the baseline hazard unspecified and assumes proportional hazards.

# Load required libraries
library(survival)
library(muhaz)
library(ggplot2)
set.seed(123)
n <- 200
t_seq <- seq(0.01, 100, by = 0.1)


### 1. Simulate Data

  # Exponential (rate = 0.1)
  exp_data <- rexp(n, rate = 0.1)
  
# Weibull (shape = 2, scale = 10)
weib_data <- rweibull(n, shape = 2, scale = 10)

# Log-normal (meanlog = 2, sdlog = 0.5)
lnorm_data <- rlnorm(n, meanlog = 2, sdlog = 0.5)

# 1. Exponential Distribution MLE 
lambda_hat_exp <- 1 / mean(exp_data)
S_exp_mle <- exp(-lambda_hat_exp * t_seq)
h_exp_mle <- rep(lambda_hat_exp, length(t_seq))
#2 Weibull Distribution MLE

# Negative log-likelihood for Weibull
nll_weibull <- function(par) {
  shape <- par[1]
  scale <- par[2]
  -sum(dweibull(weib_data, shape = shape, scale = scale, log =
                  TRUE))
}

# Optimization
weib_mle <- optim(c(1, 1), nll_weibull, method = "L-BFGS-B",
                  lower = c(0.01, 0.01))
shape_hat <- weib_mle$par[1]
scale_hat <- weib_mle$par[2]
S_weib_mle <- exp(-(t_seq / scale_hat)^shape_hat)
h_weib_mle <- (shape_hat / scale_hat) * (t_seq /
                                           scale_hat)^(shape_hat - 1)
###3. Log-normal Distribution MLE 
# Log-transform data
log_data <- log(lnorm_data)
mu_hat <- mean(log_data)
sigma_hat <- sd(log_data)
S_lnorm_mle <- 1 - plnorm(t_seq, meanlog = mu_hat, sdlog =
                            sigma_hat)
f_lnorm_mle <- dlnorm(t_seq, meanlog = mu_hat, sdlog = sigma_hat)
h_lnorm_mle <- f_lnorm_mle / S_lnorm_mle
par(mfrow = c(2, 3))


# Survival Functions
plot(t_seq, S_exp_mle, type = "l", col = "blue", main =
       "Exponential S(t)", ylab = "S(t)", xlab = "t")
plot(t_seq, S_weib_mle, type = "l", col = "red", main =
       "Weibull S(t)",ylab = "S(t)", xlab = "t")
plot(t_seq, S_lnorm_mle, type = "l", col = "green", main = 
       "lognormal S(t)",ylab = "S(t)", xlab = "t")





# Hazard Functions
plot(t_seq, h_exp_mle, type = "l", col = "blue", main =
       "Exponential h(t)", ylab = "h(t)", xlab = "t")
plot(t_seq, h_weib_mle, type = "l", col = "red", main = "Weibull
h(t)", ylab = "h(t)", xlab = "t")
plot(t_seq, h_lnorm_mle, type = "l", col = "green", main =
       "Log-normal h(t)", ylab = "h(t)", xlab = "t")
