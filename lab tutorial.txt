library(survival)
library(survminer)
data=read.csv("Cancer_Survival.csv")
head(data)
summary(data)

#(a)
km_fit=survfit(Surv(time_weeks, status) ~ treatment, data = data)


plot(km_fit,
     col = c("blue","red"),
     lwd = 2,
     xlab = "Time (Weeks)",
     ylab = "Survival Probability",
     main = "Kaplan-Meier Survival Curves")

legend("topright",
       legend = c("Treatment 0","Treatment 1"),
       col = c("blue","red"),
       lwd = 2)

# Log-rank test
logrank_test=survdiff(Surv(time_weeks, status) ~ treatment,
                         data = data)

logrank_test
p_value = 1 - pchisq(logrank_test$chisq ,df=1)
p_value

#(c)
# Exponential survival regression

exp_model <- survreg(
  Surv(time_weeks, status) ~ age + treatment +
    tumor_size + wbc_count,
  data = data,
  dist = "exponential"
)

summary(exp_model)
# Full model
full_model <- survreg(
  Surv(time_weeks, status) ~ age + treatment +
    tumor_size + wbc_count,
  data = data,
  dist = "exponential"
)

# Reduced model (without treatment)
reduced_model <- survreg(
  Surv(time_weeks, status) ~ age +
    tumor_size + wbc_count,
  data = data,
  dist = "exponential"
)

anova(reduced_model, full_model)
